Found it — playChime() schedules up to ~8.6 seconds of oscillator notes ahead of time on the Web Audio clock. Clearing the JS setTimeout stopped future chime cycles from starting, but any notes already scheduled kept playing out to completion in the background, which is why it sounded like it kept ringing after you hit the button.

Fix: every oscillator/gain pair created by playChime is now tracked in state.activeNodes. 
When you toggle chime off, stopRinging() calls a new silenceAllChimes() that immediately zeroes each gain node and force-stops each oscillator at the current audio-clock time — so pressing "Chime off" now cuts the sound instantly instead of letting the in-flight notes finish.
 

 Example codes for Chime bell ringing :
    function playChime(times) {
      const ctx = getAudioCtx();
      // Notes for a pleasing ascending chord: C5, E5, G5, C6
      const notes = [523.25, 659.25, 783.99, 1046.5];
      const count = Math.min(times, 12);
      for (let i = 0; i < count; i++) {
        const delay = i * 0.72;
        notes.forEach((freq, ni) => {
          const t = ctx.currentTime + delay + ni * 0.11;
          // Oscillator (bell tone)
          const osc = ctx.createOscillator();
          const gain = ctx.createGain();
          osc.connect(gain);
          gain.connect(ctx.destination);
          osc.type = 'sine';
          osc.frequency.setValueAtTime(freq, t);
          gain.gain.setValueAtTime(0, t);
          gain.gain.linearRampToValueAtTime(0.18, t + 0.01);
          gain.gain.exponentialRampToValueAtTime(0.0001, t + 1.6);
          osc.start(t);
          osc.stop(t + 1.8);

          // Subtle harmonic overtone
          const osc2 = ctx.createOscillator();
          const gain2 = ctx.createGain();
          osc2.connect(gain2);
          gain2.connect(ctx.destination);
          osc2.type = 'sine';
          osc2.frequency.setValueAtTime(freq * 2.756, t);
          gain2.gain.setValueAtTime(0, t);
          gain2.gain.linearRampToValueAtTime(0.05, t + 0.01);
          gain2.gain.exponentialRampToValueAtTime(0.0001, t + 0.8);
          osc2.start(t);
          osc2.stop(t + 1.0);
        });
      }
    }
